## DATABASE

Create migration:

V12__create_placement_tables.sql

Tables:

placement_profiles
placement_targets
placement_preferences

### placement_profiles

* id UUID PK
* user_id UUID NOT NULL
* domain_id BIGINT NULL
* current_cgpa DECIMAL(4,2)
* target_package_lpa DECIMAL(6,2)
* preferred_location VARCHAR(200)
* created_at TIMESTAMP WITH TIME ZONE DEFAULT now()
* updated_at TIMESTAMP WITH TIME ZONE DEFAULT now()

Constraint:

UNIQUE(user_id, domain_id)

Use existing trigger:

set_updated_at()

---

### placement_targets

* id UUID PK
* profile_id UUID NOT NULL FK → placement_profiles(id) ON DELETE CASCADE
* company_name VARCHAR(255) NOT NULL
* role_name VARCHAR(255) NOT NULL
* priority INTEGER DEFAULT 1
* created_at TIMESTAMP WITH TIME ZONE DEFAULT now()

---

### placement_preferences

* id UUID PK
* profile_id UUID NOT NULL FK → placement_profiles(id) ON DELETE CASCADE
* prefers_remote BOOLEAN DEFAULT false
* prefers_hybrid BOOLEAN DEFAULT false
* prefers_onsite BOOLEAN DEFAULT false
* preferred_company_size VARCHAR(50)
* preferred_industry VARCHAR(100)
* created_at TIMESTAMP WITH TIME ZONE DEFAULT now()
* updated_at TIMESTAMP WITH TIME ZONE DEFAULT now()

---

Indexes:

* idx_placement_profiles_user
